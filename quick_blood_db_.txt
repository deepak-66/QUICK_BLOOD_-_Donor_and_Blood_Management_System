-- Q20 NEW: Units supplied to each hospital by each blood bank

SELECT h.hospital_name, bb.bank_name, SUM(r.units_required) AS units_supplied
FROM requests r
JOIN hospitals h ON r.hospital_id = h.hospital_id
JOIN blood_banks bb ON r.bank_id = bb.bank_id

-- Trigger Demonstration 4
-- TRIGGER: trg_reduce_stock_on_complete
-- USE : completing a request reduces the bank's stock
-- Request 2 is Approved (B+, 1 unit, bank 3). Bank 3 holds 18 units of B+.

UPDATE requests
SET status = 'Completed'
WHERE request_id = 2;

SELECT units_available
FROM blood_inventory
WHERE bank_id = 3 AND blood_group_id = 3;

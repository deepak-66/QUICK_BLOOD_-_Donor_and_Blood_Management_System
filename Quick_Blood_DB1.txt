DROP DATABASE IF EXISTS quick_blood_database;
CREATE DATABASE quick_blood_database;
USE quick_blood_database;

-- =====================================================
-- TABLE 1 : BLOOD GROUPS (lookup table, new)
-- =====================================================
-- USE : Lists the 8 valid blood groups once. Other tables point to it, so no typos like a+ or AB.

CREATE TABLE blood_groups (
    blood_group_id TINYINT AUTO_INCREMENT PRIMARY KEY,
    group_name     VARCHAR(3) NOT NULL UNIQUE
);

-- =====================================================
-- TABLE 2 : HOSPITALS
-- =====================================================
-- USE : Stores hospitals that raise blood requests.

CREATE TABLE hospitals (
    hospital_id   INT AUTO_INCREMENT PRIMARY KEY,
    hospital_name VARCHAR(100) NOT NULL,
    city          VARCHAR(50)  NOT NULL,
    contact_no    VARCHAR(15),
    UNIQUE (hospital_name, city)
);

-- =====================================================
-- TABLE 3 : BLOOD BANKS
-- =====================================================
-- USE : Stores blood banks that receive donations, hold stock and fulfil requests.

CREATE TABLE blood_banks (
    bank_id    INT AUTO_INCREMENT PRIMARY KEY,
    bank_name  VARCHAR(100) NOT NULL,
    location   VARCHAR(100) NOT NULL,
    contact_no VARCHAR(15),
    UNIQUE (bank_name, location)
);

-- =====================================================
-- TABLE 4 : DONORS
-- =====================================================
-- USE : Stores people who donate blood, with their blood group and city.

CREATE TABLE donors (
    donor_id       INT AUTO_INCREMENT PRIMARY KEY,
    donor_name     VARCHAR(100) NOT NULL,
    gender         ENUM('Male','Female','Other') NOT NULL,
    blood_group_id TINYINT NOT NULL,
    age            INT NOT NULL,
    phone          VARCHAR(15) NOT NULL UNIQUE,
    email          VARCHAR(100) UNIQUE,
    city           VARCHAR(50) NOT NULL,

    CONSTRAINT chk_donor_age CHECK (age BETWEEN 18 AND 65),

    FOREIGN KEY (blood_group_id)
    REFERENCES blood_groups(blood_group_id)
);

-- =====================================================
-- TABLE 5 : RECIPIENTS
-- =====================================================
-- USE : Stores patients who need blood and the hospital they are admitted to.

CREATE TABLE recipients (
    recipient_id   INT AUTO_INCREMENT PRIMARY KEY,
    recipient_name VARCHAR(100) NOT NULL,
    blood_group_id TINYINT NOT NULL,
    age            INT NOT NULL,
    hospital_id    INT,
    phone          VARCHAR(15) NOT NULL,
    city           VARCHAR(50),

    CONSTRAINT chk_recipient_age CHECK (age BETWEEN 0 AND 120),

    FOREIGN KEY (blood_group_id)
    REFERENCES blood_groups(blood_group_id),

    FOREIGN KEY (hospital_id)
    REFERENCES hospitals(hospital_id)
    ON DELETE SET NULL
);

-- =====================================================
-- TABLE 6 : DONATIONS
-- =====================================================
-- USE : Records each donation: which donor gave to which bank, when and how many units.

CREATE TABLE donations (
    donation_id   INT AUTO_INCREMENT PRIMARY KEY,
    donor_id      INT NOT NULL,
    bank_id       INT NOT NULL,
    donation_date DATE NOT NULL,
    units_donated INT NOT NULL DEFAULT 1,

    CONSTRAINT chk_units_donated CHECK (units_donated > 0),

    FOREIGN KEY (donor_id)
    REFERENCES donors(donor_id),

    FOREIGN KEY (bank_id)
    REFERENCES blood_banks(bank_id)
);

-- =====================================================
-- TABLE 7 : BLOOD INVENTORY
-- =====================================================
-- USE : Tracks units of each blood group currently held by each blood bank.

CREATE TABLE blood_inventory (
    inventory_id    INT AUTO_INCREMENT PRIMARY KEY,
    bank_id         INT NOT NULL,
    blood_group_id  TINYINT NOT NULL,
    units_available INT NOT NULL DEFAULT 0,

    CONSTRAINT chk_units_available CHECK (units_available >= 0),
    UNIQUE (bank_id, blood_group_id),

    FOREIGN KEY (bank_id)
    REFERENCES blood_banks(bank_id)
    ON DELETE CASCADE,

    FOREIGN KEY (blood_group_id)
    REFERENCES blood_groups(blood_group_id)
);

-- =====================================================
-- TABLE 8 : REQUESTS
-- =====================================================
-- USE : Records blood requests from hospitals for recipients, and which bank fulfils them.

-- bank_id is NULL while a request is Pending.

CREATE TABLE requests (
    request_id     INT AUTO_INCREMENT PRIMARY KEY,
    recipient_id   INT NOT NULL,
    hospital_id    INT NOT NULL,
    bank_id        INT NULL,
    blood_group_id TINYINT NOT NULL,
    units_required INT NOT NULL,
    request_date   DATE NOT NULL DEFAULT (CURRENT_DATE),
    status         ENUM('Pending','Approved','Completed') NOT NULL DEFAULT 'Pending',

    CONSTRAINT chk_units_required CHECK (units_required > 0),

    FOREIGN KEY (recipient_id)
    REFERENCES recipients(recipient_id),

    FOREIGN KEY (hospital_id)
    REFERENCES hospitals(hospital_id),

    FOREIGN KEY (bank_id)
    REFERENCES blood_banks(bank_id),

    FOREIGN KEY (blood_group_id)
    REFERENCES blood_groups(blood_group_id)
);

-- PART 4 : VIEWS
-- =====================================================


-- 1.  view_donor_details

CREATE OR REPLACE VIEW view_donor_details AS
SELECT
    d.donor_id,
    d.donor_name,
    bg.group_name AS blood_group,
    d.city,
    MAX(dn.donation_date) AS last_donation_date
FROM donors d
JOIN blood_groups bg ON d.blood_group_id = bg.blood_group_id
LEFT JOIN donations dn ON d.donor_id = dn.donor_id
GROUP BY d.donor_id, d.donor_name, bg.group_name, d.city;

-- 2. view_blood_stock

CREATE OR REPLACE VIEW view_blood_stock AS
SELECT
    bb.bank_name,
    bb.location,
    bg.group_name AS blood_group,
    bi.units_available
FROM blood_inventory bi
JOIN blood_banks bb ON bi.bank_id = bb.bank_id
JOIN blood_groups bg ON bi.blood_group_id = bg.blood_group_id;

-- 3. view_donation_history

CREATE OR REPLACE VIEW view_donation_history AS
SELECT
    d.donor_name,
    bb.bank_name,
    dn.donation_date,
    dn.units_donated
FROM donations dn
JOIN donors d ON dn.donor_id = d.donor_id
JOIN blood_banks bb ON dn.bank_id = bb.bank_id;

-- 4. view_pending_requests

CREATE OR REPLACE VIEW view_pending_requests AS
SELECT
    r.request_id,
    rp.recipient_name,
    h.hospital_name,
    bg.group_name AS blood_group,
    r.units_required,
    r.status
FROM requests r
JOIN recipients rp ON r.recipient_id = rp.recipient_id
JOIN hospitals h ON r.hospital_id = h.hospital_id
JOIN blood_groups bg ON r.blood_group_id = bg.blood_group_id
WHERE r.status = 'Pending';

-- 5. view_total_donations

CREATE OR REPLACE VIEW view_total_donations AS
SELECT
    d.donor_id,
    d.donor_name,
    COUNT(dn.donation_id) AS total_donations
FROM donors d
LEFT JOIN donations dn ON d.donor_id = dn.donor_id
GROUP BY d.donor_id, d.donor_name;

-- 6. view_request_fulfilment

CREATE OR REPLACE VIEW view_request_fulfilment AS
SELECT
    r.request_id,
    h.hospital_name,
    bb.bank_name,
    bg.group_name AS blood_group,
    r.units_required,
    r.status
FROM requests r
JOIN hospitals h ON r.hospital_id = h.hospital_id
JOIN blood_groups bg ON r.blood_group_id = bg.blood_group_id
LEFT JOIN blood_banks bb ON r.bank_id = bb.bank_id;

-- PART 5 : STORED PROCEDURES
-- =====================================================

-- PROCEDURE NAME : sp_get_donor_history
-- USE            : Shows all donations made by one donor. Input: donor_id.

DROP PROCEDURE IF EXISTS sp_get_donor_history;

DELIMITER $$

CREATE PROCEDURE sp_get_donor_history
(
IN p_donor_id INT
)
BEGIN

SELECT *
FROM donations
WHERE donor_id = p_donor_id;

END $$

DELIMITER ;

-- ----------------------------------------------------

-- PROCEDURE NAME : sp_available_blood
-- USE            : Lists banks holding a blood group and their units. Input: blood group, e.g. A+.

DROP PROCEDURE IF EXISTS sp_available_blood;

DELIMITER $$

CREATE PROCEDURE sp_available_blood
(
IN p_group VARCHAR(3)
)
BEGIN

SELECT bi.inventory_id, bi.bank_id, bg.group_name AS blood_group, bi.units_available
FROM blood_inventory bi
JOIN blood_groups bg ON bi.blood_group_id = bg.blood_group_id
WHERE bg.group_name = p_group;

END $$

DELIMITER ;

-- ----------------------------------------------------

-- PROCEDURE NAME : sp_pending_requests
-- USE            : Lists all requests that are still Pending. No input.

DROP PROCEDURE IF EXISTS sp_pending_requests;

DELIMITER $$

CREATE PROCEDURE sp_pending_requests()
BEGIN

SELECT r.request_id, r.recipient_id, r.hospital_id, bg.group_name AS blood_group,
       r.units_required, r.request_date, r.status
FROM requests r
JOIN blood_groups bg ON r.blood_group_id = bg.blood_group_id
WHERE r.status = 'Pending';

END $$

DELIMITER ;

-- ----------------------------------------------------

-- PROCEDURE NAME : sp_donors_by_city
-- USE            : Lists donors living in a city. Input: city name.

DROP PROCEDURE IF EXISTS sp_donors_by_city;

DELIMITER $$

CREATE PROCEDURE sp_donors_by_city
(
IN p_city VARCHAR(50)
)
BEGIN

SELECT d.donor_id, d.donor_name, d.gender, bg.group_name AS blood_group,
       d.age, d.phone, d.email, d.city
FROM donors d
JOIN blood_groups bg ON d.blood_group_id = bg.blood_group_id
WHERE d.city = p_city;

END $$

DELIMITER ;

-- ----------------------------------------------------

-- PROCEDURE NAME : sp_bloodbank_stock
-- USE            : Shows the stock held by one blood bank. Input: bank_id.

DROP PROCEDURE IF EXISTS sp_bloodbank_stock;

DELIMITER $$

CREATE PROCEDURE sp_bloodbank_stock
(
IN p_bank INT
)
BEGIN

SELECT bi.inventory_id, bi.bank_id, bg.group_name AS blood_group, bi.units_available
FROM blood_inventory bi
JOIN blood_groups bg ON bi.blood_group_id = bg.blood_group_id
WHERE bi.bank_id = p_bank;

END $$

DELIMITER ;

-- ----------------------------------------------------
-- PROCEDURE NAME : sp_approve_request
-- USE            : Assigns a blood bank to a Pending request and marks it Approved. Input: request_id, bank_id.

DROP PROCEDURE IF EXISTS sp_approve_request;

DELIMITER $$

CREATE PROCEDURE sp_approve_request
(
IN p_request_id INT,
IN p_bank_id INT
)
BEGIN

UPDATE requests
SET bank_id = p_bank_id,
    status = 'Approved'
WHERE request_id = p_request_id
AND status = 'Pending';

END $$

DELIMITER ;

-- =====================================================
-- PART 6 : TRIGGERS
-- =====================================================

-- TABLE NAME : donation_log
-- USE        : Supporting table. Stores a log row each time a donation is inserted.

CREATE TABLE donation_log
(
    log_id INT AUTO_INCREMENT PRIMARY KEY,
    donor_id INT,
    units_donated INT,
    action_date DATETIME DEFAULT CURRENT_TIMESTAMP
);

--  ----------------------------------------------------

-- TRIGGER NAME : trg_log_donation
-- USE          : Fires AFTER INSERT on donations. Writes a row to donation_log for every donation.

DROP TRIGGER IF EXISTS trg_log_donation;

DELIMITER $$

CREATE TRIGGER trg_log_donation
AFTER INSERT
ON donations
FOR EACH ROW
BEGIN

INSERT INTO donation_log
(donor_id, units_donated)
VALUES
(NEW.donor_id, NEW.units_donated);

END $$

DELIMITER ;

-- ----------------------------------------------------
-- TRIGGER NAME : trg_add_stock_on_donation
-- USE          : Fires AFTER INSERT on donations. Adds the donated units to that bank's inventory for the donor's blood group.

DROP TRIGGER IF EXISTS trg_add_stock_on_donation;

DELIMITER $$

CREATE TRIGGER trg_add_stock_on_donation
AFTER INSERT
ON donations
FOR EACH ROW
BEGIN

INSERT INTO blood_inventory (bank_id, blood_group_id, units_available)
SELECT NEW.bank_id, d.blood_group_id, NEW.units_donated
FROM donors d
WHERE d.donor_id = NEW.donor_id
ON DUPLICATE KEY UPDATE
units_available = units_available + NEW.units_donated;

END $$

DELIMITER ;

-- ----------------------------------------------------

-- TRIGGER NAME : trg_prevent_negative_units
-- USE          : Fires BEFORE UPDATE on blood_inventory. Blocks stock from going below zero.

DROP TRIGGER IF EXISTS trg_prevent_negative_units;

DELIMITER $$

CREATE TRIGGER trg_prevent_negative_units
BEFORE UPDATE
ON blood_inventory
FOR EACH ROW
BEGIN

IF NEW.units_available < 0 THEN

SIGNAL SQLSTATE '45000'
SET MESSAGE_TEXT='Units Cannot Be Negative';

END IF;

END $$

DELIMITER ;

-- ----------------------------------------------------
-- TRIGGER NAME : trg_reduce_stock_on_complete
-- USE          : Fires AFTER UPDATE on requests. When a request becomes Completed, subtracts its units from the assigned bank's stock.

DROP TRIGGER IF EXISTS trg_reduce_stock_on_complete;

DELIMITER $$

CREATE TRIGGER trg_reduce_stock_on_complete
AFTER UPDATE
ON requests
FOR EACH ROW
BEGIN

IF NEW.status = 'Completed'
AND OLD.status <> 'Completed'
AND NEW.bank_id IS NOT NULL THEN

UPDATE blood_inventory
SET units_available = units_available - NEW.units_required
WHERE bank_id = NEW.bank_id
AND blood_group_id = NEW.blood_group_id;

END IF;

END $$

DELIMITER ;
